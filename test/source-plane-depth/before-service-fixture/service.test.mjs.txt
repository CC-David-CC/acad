// Production service, actual plane operation/classes and genuine Workers.
// Publication deliberately stops at NATIVE_NOT_RUN; no successful native model.
import test from 'node:test';import assert from 'node:assert/strict';import fs from 'node:fs';import vm from 'node:vm';import {createRequire} from 'node:module';import {randomUUID} from 'node:crypto';
import {installWindow,requests,setReplyHook,drain,lifecycle} from './platform.mjs';installWindow();const require=createRequire(import.meta.url),api=require('../../.build/source-plane-depth/service.cjs'),ts=require('typescript');
async function source(orientation='XY',depth=0){const frame=api.sketchBoundaryFrame(api.realPlane(orientation,depth)),e=await api.runSketchSolve('/static/sketch.js',{task:'sketch-transition-empty',revision:randomUUID(),unit:'millimeter',frame}),r=await api.runSketchSolve('/static/sketch.js',{task:'sketch-transition-primitive',revision:randomUUID(),sketchJSON:e.serializedSketch,externalParameters:{},primitive:{id:'agent-primitive-'+randomUUID(),kind:'Circle',role:null,stage:0,data:{c:{x:0,y:0},r:2}}});assert(r.ok);return r.serializedSketch;}
const craft=fs.readFileSync(new URL('../../web/app/cad/craft/craftBundle.ts',import.meta.url),'utf8'),tree=ts.createSourceFile('craft.ts',craft,ts.ScriptTarget.Latest,true),guards=[],coverage=[];function visit(n){if(ts.isVariableDeclaration(n)&&n.name.getText(tree)==='verifySketchFrames')guards.push(n.initializer.getText(tree));if(ts.isIfStatement(n)&&n.expression.getText(tree)==='observedFrames.size!==sourceFrames.size')coverage.push(n.getText(tree));ts.forEachChild(n,visit)}visit(tree);assert.equal(guards.length,1);assert.equal(coverage.length,1);
function verifyFrame(candidate,preparation,next){const env={sourceFrames:new Map(preparation.sketchSourceFrames.map(r=>[r.id,r.frame])),observedFrames:new Map(),models:new Set([candidate]),registry:{findFace:id=>id===candidate.face.id?candidate.face:null},history:next.history,MOpenFaceShell:api.MOpenFaceShell,sketchBoundaryFrame:api.sketchBoundaryFrame,sameSketchSaveFrame:api.sameSketchSaveFrame,getSketchBoundaries:api.getSketchBoundaries};vm.runInNewContext(ts.transpileModule('const verify='+guards[0]+';verify();'+coverage[0],{compilerOptions:{target:ts.ScriptTarget.ES2020}}).outputText,env);}
test('discovery exposes actual plane identity; guarded no-op starts no Worker or write',async()=>{const text=await source(),f=api.fixture(text),info=f.api.inspectSourcePlanes();assert.equal(info.ready,true);assert.equal(info.features.length,1);assert.deepEqual(info.features[0],{index:0,featureId:1,faceId:f.id,orientation:'XY',depth:0,canEdit:true,reason:null});const count=requests.length,result=await f.api.editSourcePlaneDepth(f.request(-0),{valid:f.valid});assert.equal(result.changed,false);assert.equal(result.authoredUndo,false);assert.equal(requests.length,count);assert.equal(f.events.submitted.length,0);assert.equal(f.events.writes.length,0);assert.equal(f.source(),text);});
test('no-op callbacks cannot invalidate owner, source, units or current plane params then dispatch',async()=>{const text=await source();for(const change of [f=>f.invalidate(),f=>f.setSource(text+' '),f=>f.ctx.projectManager.unit='inch',f=>f.ctx.craftService.modifications$.value.history[0].params.depth=5]){const f=api.fixture(text),count=requests.length;await assert.rejects(f.api.editSourcePlaneDepth(f.request(0),{valid:()=>{change(f);return true}}),e=>e.name==='AbortError');assert.equal(requests.length,count);assert.equal(f.events.submitted.length,0);assert.equal(f.events.writes.length,0);}});
test('real Worker prepares exact source/frame/history intent; real candidate frame gate passes and workspace discards',async()=>{for(const orientation of ['XY','XZ','ZY']){const text=await source(orientation,4),f=api.fixture(text,orientation,4);f.setCandidateCheck(({candidate,preparation,next})=>{const overlay=api.createCandidateSketchStorage({getAllSketches:()=>[{id:f.id}],getSketchData:()=>text,hasSketch:()=>true,readSketch:()=>api.ReadSketch(JSON.parse(text),f.id,true)},preparation.sketchOverrides,api.ReadSketch);candidate.face.setSketch(overlay.readSketch(f.id));assert.equal(candidate.face.sketchObjects.length,1);verifyFrame(candidate,preparation,next);});await assert.rejects(f.api.editSourcePlaneDepth(f.request(-7),{valid:f.valid}),{code:'NATIVE_NOT_RUN'});assert.equal(f.events.submitted.length,1);const event=f.events.submitted[0];assert.equal(event.intent.kind,'source-plane-depth');assert.equal(event.intent.beforeDepth,4);assert.equal(event.intent.afterDepth,-7);assert.equal(event.intent.before,text);assert.equal(f.events.writes.length,1);assert.equal(f.events.commits,0);assert.equal(f.events.discards,1);assert.equal(f.source(),text);assert.equal(f.ctx.craftService.modifications$.value.history[0].params.depth,4);assert(api.sameSketchSaveFrame(f.frame,api.sketchBoundaryFrame(f.model.face)));}});
test('actual frame gate rejects changed params without reframe, wrong type or missing witness',async()=>{const text=await source(),f=api.fixture(text);f.setCandidateCheck(({candidate,preparation,next})=>{const valid=preparation.sketchSourceFrames[0];assert.throws(()=>verifyFrame(candidate,{...preparation,sketchSourceFrames:[{id:f.id,frame:f.frame}]},next));assert.throws(()=>verifyFrame(candidate,preparation,{history:[{type:'UNKNOWN'}]}));assert.throws(()=>verifyFrame(candidate,{...preparation,sketchSourceFrames:[{id:'missing',frame:valid.frame}]},next));verifyFrame(candidate,preparation,next);});await assert.rejects(f.api.editSourcePlaneDepth(f.request(12)),{code:'NATIVE_NOT_RUN'});assert.equal(f.events.commits,0);assert.equal(f.source(),text);});
test('stale/unknown/malformed/unsupported requests refuse before Worker or publication',async()=>{const text=await source();for(const mutate of [a=>a.revision='stale',a=>a.featureId=2,a=>a.faceId='unknown',a=>a.depth=Infinity,a=>a.extra=1]){const f=api.fixture(text),a=f.request(12),count=requests.length;mutate(a);await assert.rejects(f.api.editSourcePlaneDepth(a));assert.equal(requests.length,count);assert.equal(f.events.submitted.length,0);}const f=api.fixture(text);f.ctx.craftService.modifications$.value.history[0].params.datum={id:'unsupported'};assert.equal(f.api.inspectSourcePlanes().features[0].canEdit,false);await assert.rejects(f.api.editSourcePlaneDepth(f.request(12)),{code:'UNSUPPORTED_PLANE'});});
test('genuine held Worker replies are refused after source ABA and actual document lifetime abort',async()=>{const text=await source();for(const mode of ['aba','document']){const f=api.fixture(text);let release,ready;const held=new Promise(r=>ready=r);setReplyHook(async(value,req)=>{if(req.task==='sketch-source-plane-depth'&&value.ok){ready(value);await new Promise(r=>release=r)}});try{const p=f.api.editSourcePlaneDepth(f.request(12));const genuine=await held;assert(genuine.ok);if(mode==='aba'){f.setSource(text+' ');f.setSource(text)}else f.close();release();await assert.rejects(p,e=>e.name==='AbortError');assert.equal(f.events.submitted.length,0);assert.equal(f.events.writes.length,0);assert.equal(f.source(),text);}finally{setReplyHook(null)}}});
test('physically releases every production Worker',async()=>{await drain();assert.equal(lifecycle.started,lifecycle.terminated);});
