// Genuine production sketch Worker/client under strict CSP. No app, accounts,
// native modeling, successful publication, canvas or shared browser is used.
const fs=require('node:fs'),path=require('node:path'),os=require('node:os'),http=require('node:http'),crypto=require('node:crypto'),assert=require('node:assert/strict');
const root=path.resolve(__dirname,'../..'),out=path.join(root,'.build/source-plane-depth'),sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const puppeteer=require(process.env.PUPPETEER_CORE_PATH||'/tmp/codi-occt-release-control/node_modules/puppeteer-core');
async function main(){
 const {installWindow}=await import('./platform.mjs');installWindow();const actual=require(path.join(out,'service.cjs'));
 const frames=Object.fromEntries(['XY','XZ','ZY'].map(orientation=>[orientation,Object.fromEntries([4,-7,0,15].map(depth=>[depth,actual.sketchBoundaryFrame(actual.realPlane(orientation,depth))]))]));
 const files={'/browser.js':path.join(out,'browser.js'),'/static/sketch.js':path.join(out,'sketch/worker.js'),'/lib/verb.js':path.join(root,'web/lib/verb.js')},bodies=Object.fromEntries(Object.entries(files).map(([url,file])=>[url,fs.readFileSync(file)])),hashes=Object.fromEntries(Object.entries(bodies).map(([url,body])=>[url,sha(body)])),served=[],errors=[],record={scope:'strict-CSP production sketch Worker/client component; no native or app publication',status:'RUNNING',buildPins:JSON.parse(fs.readFileSync(path.join(out,'pins.json'))),servedHashes:hashes};
 const server=http.createServer((req,res)=>{res.setHeader('Content-Security-Policy',"default-src 'none'; script-src 'self'; worker-src 'self'; connect-src 'none'; object-src 'none'; base-uri 'none'");res.setHeader('Cache-Control','no-store');if(req.url==='/'){res.setHeader('Content-Type','text/html');res.end('<!doctype html><title>Source plane Worker component</title><script src="/browser.js"></script>');return}const body=bodies[req.url];if(!body){res.writeHead(404).end();return}served.push(req.url);res.setHeader('Content-Type','application/javascript');res.end(body)});
 await new Promise(resolve=>server.listen(0,'127.0.0.1',resolve));const profile=fs.mkdtempSync(path.join(os.tmpdir(),'codi-source-plane-depth-chrome-'));let browser;
 try{
  browser=await puppeteer.launch({executablePath:process.env.CHROME_BIN||'/snap/bin/chromium',headless:true,userDataDir:profile,args:['--no-sandbox','--disable-dev-shm-usage','--disable-background-networking','--no-first-run']});const page=await browser.newPage();let created=0,destroyed=0;page.on('workercreated',()=>created++);page.on('workerdestroyed',()=>destroyed++);page.on('pageerror',error=>errors.push(String(error)));await page.goto('http://127.0.0.1:'+server.address().port+'/',{waitUntil:'load'});
  record.observations=await page.evaluate(async frames=>{
   const api=window.SourcePlaneDepthComponent,url='/static/sketch.js',rows=[],check=(yes,message)=>{if(!yes)throw Error(message)},strip=text=>{const data=JSON.parse(text);delete data.metadata.boundaryFrame;return JSON.stringify(data)},request=(text,frame,orientation,beforeDepth,afterDepth,candidateJSON=null)=>({revision:crypto.randomUUID(),unit:'millimeter',sketchJSON:text,frame,orientation,beforeDepth,afterDepth,candidateJSON});
   for(const orientation of ['XY','XZ','ZY']){
    const frame=frames[orientation][4],empty=await api.runSketchSolve(url,{task:'sketch-transition-empty',revision:crypto.randomUUID(),unit:'millimeter',frame});check(empty.ok,'Real empty source refused');const primitive=await api.runSketchSolve(url,{task:'sketch-transition-primitive',revision:crypto.randomUUID(),sketchJSON:empty.serializedSketch,externalParameters:{},primitive:{id:'agent-primitive-'+crypto.randomUUID(),kind:'Circle',role:null,stage:0,data:{c:{x:0,y:0},r:2}}});check(primitive.ok,'Real Circle source refused');
    for(const [kind,source] of [['empty',empty.serializedSketch],['circle',primitive.serializedSketch]])for(const depth of [-7,0,15]){const r=request(source,frame,orientation,4,depth),captured=JSON.stringify(r),result=await api.runSourcePlaneDepthPreparation(url,r);check(JSON.stringify(r)===captured,'Caller input changed');check(strip(source)===strip(result.serializedSketch),'Local source intent changed');check(JSON.stringify(result.frame)===JSON.stringify(frames[orientation][depth]),'Target frame differs from real principal plane');rows.push({orientation,kind,depth,before:source,after:result.serializedSketch,frame:result.frame});}
    const original=JSON.stringify(JSON.parse(primitive.serializedSketch),null,2),moved=await api.runSourcePlaneDepthPreparation(url,request(original,frame,orientation,4,15)),restored=await api.runSourcePlaneDepthPreparation(url,request(moved.serializedSketch,moved.frame,orientation,15,4,original));check(restored.serializedSketch===original,'Historical source bytes not restored');rows.push({orientation,kind:'exact-historical-travel',before:original,after:restored.serializedSketch});
    const altered=JSON.parse(moved.serializedSketch);altered.objects[0].data.r=3;let code=null;try{await api.runSourcePlaneDepthPreparation(url,request(original,frame,orientation,4,15,JSON.stringify(altered)))}catch(error){code=error.code}check(code==='INTENT_CHANGED','Altered local intent was accepted');rows.push({orientation,kind:'candidate-local-intent-refusal',code});
   }
   const controller=new AbortController();controller.abort();let aborted=false;try{await api.runSourcePlaneDepthPreparation(url,request(rows[0].before,frames.XY[4],'XY',4,5),{signal:controller.signal})}catch(error){aborted=error.name==='AbortError'}check(aborted,'Cancelled transition accepted');return rows;
  },frames);
  await page.waitForFunction(()=>true);assert.equal(errors.length,0,errors.join('\n'));assert.equal(page.workers().length,0);assert.equal(record.observations.length,24);assert.equal(created,57);record.workers={created,destroyed,active:page.workers().length};record.chrome=await browser.version();record.served=Object.fromEntries(Object.keys(files).map(url=>[url,served.filter(row=>row===url).length]));for(const [url,file]of Object.entries(files))assert.equal(sha(fs.readFileSync(file)),hashes[url]);record.status='PASS';
 }catch(error){record.status='FAIL';record.error={name:error.name,message:error.message,stack:error.stack};throw error}finally{
  if(browser)await browser.close();await new Promise(resolve=>server.close(resolve));fs.rmSync(profile,{recursive:true,force:true});record.cleanup={ownBrowserClosed:true,ownServerClosed:true,ownProfileRemoved:!fs.existsSync(profile),sharedBrowsersUntouched:true};fs.writeFileSync(path.join(out,'browser-receipt.json'),JSON.stringify(record,null,2)+'\n');
 }
 console.log('PASS24 actual Chromium Worker observations; no native/shared publication claim');
}
main().catch(error=>{console.error(error);process.exitCode=1});
