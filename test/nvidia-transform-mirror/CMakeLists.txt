cmake_minimum_required(VERSION 3.20)
project(codi_transform_mirror_probe CXX)
if(NOT EMSCRIPTEN OR NOT DEFINED SMLIB_SOURCE_ROOT)
  message(FATAL_ERROR "Use Emscripten and -DSMLIB_SOURCE_ROOT=/path/to/usd-brep")
endif()
set(CMAKE_CXX_STANDARD 17)
file(GLOB SMLIB_SOURCES "${SMLIB_SOURCE_ROOT}/source/SMLib/src/*.cpp")
file(GLOB API_SOURCES "${SMLIB_SOURCE_ROOT}/source/SM_API/src/*.cpp")
list(FILTER SMLIB_SOURCES EXCLUDE REGEX "/StdAfx\\.cpp$")
list(FILTER API_SOURCES EXCLUDE REGEX "/StdAfx\\.cpp$")
add_library(smlib STATIC ${SMLIB_SOURCES})
add_library(smapi STATIC ${API_SOURCES})
target_include_directories(smlib PRIVATE "${SMLIB_SOURCE_ROOT}/source/SMLib/inc")
target_include_directories(smapi PRIVATE "${SMLIB_SOURCE_ROOT}/source/SMLib/inc" "${SMLIB_SOURCE_ROOT}/source/SM_API/inc")
foreach(target smlib smapi)
  target_compile_definitions(${target} PRIVATE NDEBUG SM_API_EXPORTS)
  target_compile_options(${target} PRIVATE -O2 -fwasm-exceptions -frtti)
endforeach()
foreach(target probe extended)
  add_executable(${target} ${target}.cpp)
  target_include_directories(${target} PRIVATE "${SMLIB_SOURCE_ROOT}/source/SMLib/inc" "${SMLIB_SOURCE_ROOT}/source/SM_API/inc")
  target_compile_definitions(${target} PRIVATE NDEBUG SM_API_EXPORTS)
  target_compile_options(${target} PRIVATE -O2 -fwasm-exceptions -frtti)
  target_link_libraries(${target} PRIVATE smapi smlib)
  target_link_options(${target} PRIVATE -O2 -fwasm-exceptions -sDYNAMIC_EXECUTION=0 -sENVIRONMENT=node -sMODULARIZE=1 -sEXPORT_ES6=1 -sFILESYSTEM=0 -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=32MB -sMAXIMUM_MEMORY=4GB -sSTACK_SIZE=4MB "-sEXPORTED_FUNCTIONS=['_run_${target}']")
  set_target_properties(${target} PROPERTIES SUFFIX ".mjs")
endforeach()
